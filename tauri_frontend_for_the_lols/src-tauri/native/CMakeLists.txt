cmake_minimum_required(VERSION 3.18)
project(tnr_tauri_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(TNRP_USE_QT OFF CACHE BOOL "" FORCE)
set(TNRP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(WIN32)
  add_compile_definitions(WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
endif()
add_subdirectory(../../../protocol_parser_library tnrp_build)
add_library(tnr_tauri_core STATIC core.cpp)
target_link_libraries(tnr_tauri_core PRIVATE tnrp)
set_target_properties(xlsxwriter PROPERTIES OUTPUT_NAME tnr_xlsx DEBUG_POSTFIX "")
set_target_properties(libzstd_static PROPERTIES OUTPUT_NAME tnr_zstd DEBUG_POSTFIX "")
set_target_properties(zlibstatic PROPERTIES OUTPUT_NAME tnr_zlib DEBUG_POSTFIX "")
install(TARGETS tnr_tauri_core tnrp xlsxwriter libzstd_static zlibstatic ARCHIVE DESTINATION lib)
